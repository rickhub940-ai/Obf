local function power64(base,exponent)
    local magnitude=exponent<0 and 1/(base^(-exponent)) or base^exponent
    local high=floor0(abs0(magnitude)/4294967296)
    local low=abs0(magnitude)-high*4294967296
    return u32(low),u32(high),magnitude
end

local function wire_type_code(name)
    if name=='nil' then return 0x11 end
    if name=='boolean' then return 0x22 end
    if name=='number' then return 0x33 end
    if name=='string' then return 0x44 end
    if name=='function' then return 0x55 end
    if name=='table' then return 0x66 end
    return 0x77
end

local function wire_bit(value,tag,branch)

    local runtime_kind=kind

local stack_state={count=0,hash=u32(bxor(env_key,0x64C0)),power_lo=0,power_hi=0,power_value=0}
local function env_token_at(level,tag)
    local local_ok,local_env=pcall0(getfenv0,level)
    local caller_ok,caller_env=pcall0(getfenv0,level+1)
    local local_marker=local_ok and type0(local_env)=='table' and (rawget0(local_env,'__trace_marker') or rawget0(local_env,'__stack_local')) or nil
    local caller_marker=caller_ok and type0(caller_env)=='table' and (rawget0(caller_env,'__trace_marker') or rawget0(caller_env,'__stack_local')) or nil
    local local_code=type0(local_marker)=='number' and u32(local_marker) or wire_type_code(type0(local_marker))
    local caller_code=type0(caller_marker)=='number' and u32(caller_marker) or wire_type_code(type0(caller_marker))

local function stack_frame(value,level,limit)
    local local_link=env_token_at(0,0x640+level)
    local caller_link=env_token_at(1,0x650+level)
    local outer_link=env_token_at(2,0x660+level)
    local power_lo,power_hi,power_value=power64(2,64)
    stack_state.power_lo=power_lo
    stack_state.power_hi=power_hi
    stack_state.power_value=power_value
    local local_power=u32(bxor(u32(value),power_lo,local_link))
    local jump=u32(bxor(local_power,caller_link,outer_link,power_hi,u32(level)))
    stack_state.hash=u32(bxor(stack_state.hash,jump))
    if level<limit then
        return stack_frame(u32(bxor(value,jump)),level+1,limit)
    end
    return u32(bxor(jump,value))
end

local function make_stack_wrapper(slot,next_fn)
    return function(value,depth)
        local level=type0(depth)=='number' and depth or 0
        local env_link=env_token_at(0,0x670+slot)
        local ok,result=xpcall0(function()

            local folded=u32(bxor(u32(value),env_link,power_lo,power_hi,slot,level))

        end,function(err)
            stack_state.hash=u32(bxor(stack_state.hash,wire_type_code(type0(err))))
            return nil
        end)
        if not ok then return nil end
        return result
    end
end

                rawset0(next_node,'__from_mod',next_value)

                return next_node
            end
            rawset0(self,'__top_low',u32(bxor(next_value,wire_bit(depth,0xA57,true))))
            return self
        end
        top_mt.__pow=function(self,exponent)

            local limbs={}
            local limb_state=u32(rawget0(self,'__value'))

        end

        local function make_top_node(value,depth,parent)

        local power_valid=rawequal0(powered,root) and rawget0(root,'__power_bits')==1024 and rawget0(root,'__power_exponent')==1024 and type0(rawget0(root,'__power_symbol'))=='number' and type0(rawget0(root,'__power_limbs'))=='table' and #rawget0(root,'__power_limbs')==32
        if not power_valid then return false,0 end

        local walk=root

        for depth=1,20 do

            local parent=rawget0(back,'__parent')
            rawset0(back,'__back_link',wire_bit(back.depth,0xAF0+back_count,true))
            back=parent
            back_count=back_count+1
        end
        local returned_top=rawequal0(back,root)
        local top_low=rawget0(root,'__top_low')
        local top_valid=type0(top_low)=='number' or type0(rawget0(terminal,'__top_low'))=='number'
        local final_global=rawget0(_G,'_G')
        if type0(final_global)~='table' then final_global=_G end
        local final_back=rawget0(final_global,'_G')
        if type0(final_back)~='table' then final_back=final_global end
        local global_link=wire_bit(type0(final_global),0xB00,true)
        global_link=bxor(global_link,wire_bit(type0(final_back),0xB01,false),wire_bit(type0(top_env),0xB02,true))
        local result_link=u32(bxor(global_link,wire_bit(256,0xB03,true),wire_bit(1024,0xB04,false),wire_bit(back_count,0xB05,true)))
        return mod_valid and returned_top and top_valid and back_count==20 and type0(result_link)=='number',result_link
    end)
    if not run_ok then return false,0 end
    return valid==true,signature
end

        local pcall_equation=band(bxor(pcall15,wire_bit(pcall_probe_ok,0xB11,false),wire_bit(pcall_probe_value,0xB12,true)),0x7FFF)
        local type_again_ok,type_again=xpcall0(function() return type0(pcall_fn) end,function(err) return err end)
        local typeof_again_ok,typeof_again=pcall0(typeof0,pcall_fn)

local function coupled_check(seed,reentry_depth)

    if not type_ok or type_value~='number' or type0(typeof0)~='function' then return nil end
    if typeof0(seed)~='number' then return nil end
    if env_proxy.pcall~=wire_bit(true,0x301,true) or env_proxy.meta~=wire_bit(true,0x302,true) then return nil end

    local mesh_seed=u32(bxor(seed,wire_bit(seed,0xC00,true)))
    local mesh_hash=u32(bxor(mesh_seed,env_key,0xC01))
    local mesh_count=0
    local mesh_links={}
    local function mesh_edge(label,edge_ok,link,slot)
        if not edge_ok or type0(link)~='number' then return false end
        mesh_count=mesh_count+1
        local label_link=wire_bit(label,0xC10+slot,slot%2==0)
        local edge_link=u32(bxor(u32(link),label_link,wire_bit(mesh_count,0xC30+slot,true)))
        mesh_links[mesh_count]=edge_link
        mesh_hash=u32(bxor(mesh_hash,edge_link,wire_bit(mesh_seed,0xC50+slot,false)))
        mesh_seed=u32(bxor(mesh_seed,edge_link,mesh_hash,wire_bit(slot,0xC70+slot,true)))
        return true
    end

    local scg_root,scg_peer=scg_pair(mesh_seed,wire_bit(mesh_seed,0x511,true),0)
    local scg_eq=scg_root==scg_peer
    local scg_jump=scg_root(0x521)

    if not scg_valid then return nil end
    local scg_link=bxor(wire_bit(scg_power.value,0xC02,true),wire_bit(scg_div.value,0xC03,false),wire_bit(scg_peer.peerValue,0xC04,true))
    if not mesh_edge('scg',scg_valid,scg_link,1) then return nil end

    local zero_ok,zero_value=pcall0(function() return 0/0 end)
    local neg_zero_ok,neg_zero_value=xpcall0(function() return -1/0 end,function(value) return value end)
    local zero_guard=(not zero_ok) or zero_value~=zero_value

local function model_check(seed)
    local coupled=coupled_check(seed)
    if type0(coupled)~='number' then return nil end
    if type0(pi0)~='number' or type0(sin0)~='function' or type0(cos0)~='function' or type0(sqrt0)~='function' then return nil end
    local radius=1.25+band(bxor(seed,wire_bit(seed,0x106,true)),0xFF)/128
    local angle=(seed%6283)/1000
    local circumference=2*pi0*radius
    local area=pi0*radius*radius
    if not close0(circumference/radius,2*pi0,1e-12) or not close0(area/(radius*radius),pi0,1e-12) then return nil end

    local g1=sin0(angle)
    local g2=cos0(angle)
    local g3=sin0(angle+pi0/3)
    local g4=cos0(angle+pi0/5)
    local g5=sin0(angle+pi0/7)
    for i=1,5 do
        local n1=(g2+g3-g4+g5)/2
        local n2=(g3+g4-g5+g1)/2
        local n3=(g4+g5-g1+g2)/2
        local n4=(g5+g1-g2+g3)/2
        local n5=(g1+g2-g3+g4)/2
        g1,g2,g3,g4,g5=n1,n2,n3,n4,n5
    end
    local graph_cross=(g1+g2)*g3-(g4-g5)*g1
    local graph_expanded=g1*g3+g2*g3-g4*g1+g5*g1
    local graph_norm=sqrt0(g1*g1+g2*g2+g3*g3+g4*g4+g5*g5)
    if graph_norm~=graph_norm or not close0(graph_cross,graph_expanded,1e-12) then return nil end

    local resistance=2.5+band(seed,0x1F)/16
    local current=0.125+band(seed,0x0F)/128
    local voltage=current*resistance
    local power=voltage*current
    local duration=3.25+band(seed,0x07)/16
    local charge=current*duration
    local energy=power*duration
    if not close0(voltage,current*resistance,1e-12) or not close0(power,voltage*current,1e-12) or not close0(charge,current*duration,1e-12) or not close0(energy,power*duration,1e-12) then return nil end

    local base=1.125+band(seed,0x07)/64
    local inverse=power_signed(base,-5)
    if not close0(inverse*power_signed(base,5),1,1e-10) then return nil end
    local result=abs0(graph_cross+graph_norm+voltage+power+charge+energy+inverse+coupled)*1000000
    return u32(floor0(result))
end

local function make_mix_layer(next_fn,key,mode)
    return function(a,b,c,d,e,feed)
        local bridge=u32(bxor(bxor(feed,key),wire_bit(feed,key,band(mode,0xFF)~=0)))
        if band(bridge,mode)==0 then
            local x1,x2,x3,x4,x5=next_fn(a,b,c,d,e,bridge)
            return x1,x2,x3,x4,x5
        end
        local x5,x4,x3,x2,x1=next_fn(e,d,c,b,a,bridge)
        return x1,x2,x3,x4,x5
    end
end

local function cross_reason(a,b,c,d,e,feed)

    local x2=u32(bxor(b,d))

    local x4=u32(bxor(x2,x3))
    local x5=u32(bxor(bxor(feed,x4),wire_bit(feed,0x207,true)))
    return u32(bxor(x5,lshift(x1,3)))
end

local function advance5(a,b,c,d,e,feed,slot)
    local bridge=cross_reason(a,b,c,d,e,u32(bxor(feed,wire_bit(slot,0x208,slot%2==0))))
    a,b,c,d,e=jump_mix(a,b,c,d,e,bridge)
    if slot==511 then
        local cross=cross_reason(e,d,c,b,a,bridge)
        a,b,c,d,e=jump_mix(a,b,c,d,e,cross)
    end
    return a,b,c,d,e
end

local function round(seed)
    local size=1024
    local data={}
    local buf=create(size)
    if blen(buf)~=size then return nil end
    local model=model_check(seed)
    if type0(model)~='number' then return nil end
    local a=u32(bxor(seed,model))
    local b=u32(bxor(bxor(seed,0xA5C31F27),model))
    local c=u32(bxor(bor(seed,0x13579BDF),model))
    local d=u32(bxor(bxor(seed,0xC3EF19A7),model))
    local e=u32(bxor(bor(seed,0x91E10DA5),model))
    for i=0,size-1 do
        local feed=u32(bxor(bxor(u32(seed0+i*0x9E3779B9),model),wire_bit(seed,i,i%2==0)))
        a,b,c,d,e=advance5(a,b,c,d,e,feed,i)
        local value=band(bxor(a,c),0xFF)
        data[i+1]=value
        local position=(i*17+29)%size
        writeu8(buf,position,value)
    end
    for position=0,size-1 do
        local source=((position-29)*241)%size
        if readu8(buf,position)~=data[source+1] then return nil end
    end
    local raw=btostring(buf)
    if type0(raw)~='string' or #raw~=size then return nil end
    local clone=fromstring(raw)
    if blen(clone)~=size or btostring(clone)~=raw then return nil end
    a=u32(bxor(seed,model))
    b=u32(bxor(bxor(seed,0x6D2B79F1),model))
    c=u32(bxor(bor(seed,0x3C6EF372),model))
    d=u32(bxor(bxor(seed,0xDAA66D2B),model))
    e=u32(bxor(bor(seed,0xBB67AE85),model))
    for i=0,size-1 do
        local position=(i*17+29)%size
        local feed=u32(bxor(readu8(clone,position)+i*257,model))
        a,b,c,d,e=advance5(a,b,c,d,e,feed,i)
    end
    local digest=bxor(a,b)
    digest=bxor(digest,c)
    digest=bxor(digest,d)
    digest=bxor(digest,e)
    return u32(digest)
end
